Сколько нулей (цифр) в записи числа

НАИЗУСТЬ
n = 16 ** 820 - 2 ** 761 + 14
k = 0      # в k считаем количество нулей
q = 4      # основание системы счисления
while n != 0:
    d = n % q      # цифра в текущем разряде
    n = n // q     # понижаем разряд
    if d == 0:     # проверяем, что цифра 0
        k += 1
print(k)           # 378

НАИЗУСТЬ
res = []    # количество нулей для каждого x
for x in range(1, 2031):     # x не превышает 2030
    n = 7 ** 170 + 7 ** 100 - x
    k = 0
    q = 7
    while n != 0:
        d = n % q
        n = n // q
        if d == 0:
            k += 1
    res.append(k)
print(max(res))              # 73
